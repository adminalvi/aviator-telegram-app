<script>

const tg = window.Telegram.WebApp;

tg.ready();

tg.expand();


const user = tg.initDataUnsafe?.user;

document.getElementById('userInfo').innerText = user ? `Player: ${user.first_name}` : 'Player: Guest';


const socket = io(); // Automatically connects to current hosted domain


let currentMultiplier = 1.00;

let isFlying = false;


const config = {

type: Phaser.AUTO,

width: window.innerWidth,

height: window.innerHeight * 0.7,

parent: 'game-container',

backgroundColor: 0x0f131c,

scene: { preload: preload, create: create, update: update }

};


const game = new Phaser.Game(config);

let plane, curveGraphics, multiplierText, statusText;

let startX = 40, startY, targetX, targetY;


function preload() {

const planeSvg = 'data:image/svg+xml;utf8,<svg xmlns="http://www.w3.org/2000/svg" width="50" height="30" viewBox="0 0 50 30"><path fill="%23e53935" d="M2,15 L15,10 L35,2 L48,15 L35,28 L15,20 Z"/></svg>';

this.load.image('plane', planeSvg);

}


function create() {

const width = this.scale.width;

const height = this.scale.height;

startY = height - 40;

targetX = startX;

targetY = startY;


curveGraphics = this.add.graphics();

plane = this.add.image(startX, startY, 'plane').setVisible(false);


multiplierText = this.add.text(width / 2, height / 2 - 20, '1.00x', {

fontSize: '48px', fontWeight: 'bold', fill: '#ffffff'

}).setOrigin(0.5);


statusText = this.add.text(width / 2, height / 2 + 30, 'WAITING...', {

fontSize: '14px', fill: '#848d9c'

}).setOrigin(0.5);


socket.on('tick', (data) => {

if (!isFlying) startFlight();

currentMultiplier = parseFloat(data.multiplier);

multiplierText.setText(currentMultiplier.toFixed(2) + 'x');

updatePlanePosition(width, height);

});


socket.on('crashed', (data) => {

triggerCrash(data.crashPoint);

});


socket.on('game_state', (data) => {

if(data.status === 'WAITING') resetCanvas();

});

}


function update() {

if (isFlying) {

curveGraphics.clear();

curveGraphics.lineStyle(3, 0xe53935, 1);

curveGraphics.beginPath();

curveGraphics.addQuadraticCurve(startX, startY, startX + (targetX - startX) * 0.2, startY, targetX, targetY);

curveGraphics.strokePath();

plane.setPosition(targetX, targetY);

}

}


function updatePlanePosition(width, height) {

let progress = Math.min((currentMultiplier - 1) / 5, 1);

targetX = startX + (progress * (width - 80));

targetY = startY - (progress * (height - 80));

}


function startFlight() {

isFlying = true;

plane.setVisible(true);

statusText.setText('');

if (tg.HapticFeedback) tg.HapticFeedback.impactOccurred('medium');

}


function triggerCrash(crashPoint) {

isFlying = false;

plane.setVisible(false);

multiplierText.setText('FLEW AWAY!').setFill('#e53935');

statusText.setText('CRASHED AT ' + parseFloat(crashPoint).toFixed(2) + 'x').setFill('#e53935');

if (tg.HapticFeedback) tg.HapticFeedback.notificationOccurred('error');

curveGraphics.clear();

}


function resetCanvas() {

isFlying = false;

targetX = startX; targetY = startY;

plane.setPosition(startX, startY).setVisible(false);

curveGraphics.clear();

multiplierText.setText('1.00x').setFill('#ffffff');

statusText.setText('WAITING...').setFill('#848d9c');

}


function triggerCashout() {

if (isFlying) {

socket.emit('player_cashout', { multiplier: currentMultiplier });

if (tg.HapticFeedback) tg.HapticFeedback.notificationOccurred('success');

tg.showAlert(`CASHED OUT AT ${currentMultiplier.toFixed(2)}x`);

}

}
